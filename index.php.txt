function refreshManualSlots(){
  var dateVal=$('mb-date').value||todayStr();

  $('mb-slot').innerHTML=slots.map(function(s){return '<option value="'+s.label+'" data-key="'+s.key+'">'+s.label+' • '+s.tag+'</option>'}).join('');
  if(slots.some(function(s){return s.label===cur}))$('mb-slot').value=cur;
}
function calcManualTotal(){
  var p=get(SK.PRICING),court=$('mb-court').value;
  var sel=$('mb-slot').selectedOptions[0];
  var key=sel?sel.getAttribute('data-key'):'morning';
  var rack=+$('mb-racket').value;
  var total=((p.courts[court]||{})[key]||0)+rack;
  $('mbTotal').textContent=fmt(total);
  return total;
}

      +'<td>'+x.capacity+' pax</td>'
      +'<td><span class="badge '+(x.status==='active'?'b-confirmed':'b-cancelled')+'">'+x.status+'</span></td>'
      +'<td><div class="tbl-actions" style="justify-content:flex-end">'

}

}

function renderReports(){
  var books=get(SK.BOOKINGS).filter(function(b){return b.status!=='cancelled'});
  var total=books.reduce(function(s,b){return s+b.total},0);
  $('rpAov').textContent=books.length?Math.round(total/books.length/1000):0;
  var fc={};books.forEach(function(b){fc[b.courtName]=(fc[b.courtName]||0)+1});
  $('topFac').innerHTML=Object.entries(fc).sort(function(a,b){return b[1]-a[1]}).slice(0,5).map(function(e,i){return '<div class="rank-item"><div class="rank-num">'+(i+1)+'</div><div class="rank-name">'+e[0]+'</div><div class="rank-val">'+e[1]+' booking</div></div>'}).join('')||'<div class="empty">Belum ada data</div>';
  var sc={};books.forEach(function(b){sc[b.slot]=(sc[b.slot]||0)+1});
  $('topSlot').innerHTML=Object.entries(sc).sort(function(a,b){return b[1]-a[1]}).slice(0,5).map(function(e,i){return '<div class="rank-item"><div class="rank-num">'+(i+1)+'</div><div class="rank-name">'+e[0]+'</div><div class="rank-val">'+e[1]+' sesi</div></div>'}).join('')||'<div class="empty">Belum ada data</div>';
  if(chartReady($('trendChart'))){
    var labels=[],data=[];
    for(var i=6;i>=0;i--){var d=new Date();d.setDate(d.getDate()-i);labels.push(d.toLocaleDateString('id-ID',{weekday:'short'}));data.push(books.filter(function(b){return parseDate(b.date).toDateString()===d.toDateString()}).length)}
    if(trendChart)trendChart.destroy();
    trendChart=new Chart($('trendChart'),{type:'bar',data:{labels:labels,datasets:[{label:'Booking',data:data,backgroundColor:'#7b1113',hoverBackgroundColor:'#b3872e',borderRadius:8,borderSkipped:false}]},
      options:{responsive:true,maintainAspectRatio:false,plugins:{legend:{display:false}},scales:{y:{ticks:{color:tickColor,font:chartFont},grid:{color:gridColor}},x:{ticks:{color:tickColor,font:chartFont},grid:{display:false}}}}});
  }
}

  var csv=[['ID','Nama','Fasilitas','Tanggal','Jam','Pemain','Metode','Status Bayar','Total','Status']].concat(get(SK.BOOKINGS).map(function(b){return [b.id,b.name,b.courtName,b.date,b.slot,b.players,b.method,b.payStatus,b.total,b.status]}))

}
